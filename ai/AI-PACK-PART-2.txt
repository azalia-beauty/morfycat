===== scripts/add-module.mjs =====
// «Докинуть код»: берёт ответ ИИ (из файла, буфера обмена или stdin), находит в нём файлы и кладёт куда нужно.
//
//   node scripts/add-module.mjs ответ.txt        из файла
//   node scripts/add-module.mjs --clip           из буфера обмена
//   node scripts/add-module.mjs -                из stdin
//
// Что понимает:
//  • один блок кода с export default { id, … }  → src/modules/<id>.js
//  • один блок кода с module.exports { handlers } → electron/modules/<имя>.cjs
//  • несколько блоков, у каждого первой строкой «// FILE: путь/к/файлу» → кладёт по указанным путям
// Перед записью проверяет синтаксис; существующие файлы сохраняет в .ghost-backup/.
import { execFileSync, spawnSync } from 'node:child_process';
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';

const root = process.cwd();
const args = process.argv.slice(2);
const dry = args.includes('--dry');
const src = args.find((a) => !a.startsWith('--') || a === '-');
const clip = args.includes('--clip');

const fail = (m) => { console.error('\n✗ ' + m + '\n'); process.exit(1); };

function readClipboard() {
  const tries = process.platform === 'win32'
    ? [['powershell', ['-NoProfile', '-Command', '[Console]::OutputEncoding=[Text.Encoding]::UTF8; Get-Clipboard -Raw']]]
    : process.platform === 'darwin' ? [['pbpaste', []]] : [['wl-paste', []], ['xclip', ['-selection', 'clipboard', '-o']], ['xsel', ['-b']]];
  for (const [cmd, a] of tries) {
    const r = spawnSync(cmd, a, { encoding: 'utf8' });
    if (r.status === 0 && r.stdout) return r.stdout;
  }
  return '';
}

function readInput() {
  if (clip) return readClipboard();
  if (src === '-') return fs.readFileSync(0, 'utf8');
  if (src) { if (!fs.existsSync(src)) fail('Файл не найден: ' + src); return fs.readFileSync(src, 'utf8'); }
  fail('Укажите файл с ответом ИИ, или --clip (буфер обмена), или - (stdin).');
}

const text = readInput().replace(/^﻿/, '');
if (!text.trim()) fail('Пусто: нечего добавлять. Скопируйте ответ ИИ и повторите.');

// — блоки кода
let blocks = [...text.matchAll(/```[\w+-]*[ \t]*\r?\n([\s\S]*?)```/g)].map((m) => m[1]);
if (!blocks.length) {
  const open = /```[\w+-]*[ \t]*\r?\n([\s\S]*)$/.exec(text);
  blocks = [open ? open[1] : text];
}

const ALLOWED = [/^src\/[\w./-]+\.(js|css)$/, /^electron\/[\w./-]+\.cjs$/, /^scripts\/[\w./-]+\.(mjs|sh|ps1)$/, /^android-plugin\/[\w./-]+$/];
const FILE_RE = /^\s*(?:\/\/|#|\/\*|<!--)\s*FILE:\s*(\S+)/i;

const plan = [];
for (const raw of blocks) {
  const code = raw.replace(/\r\n/g, '\n').trim() + '\n';
  const m = FILE_RE.exec(code.split('\n')[0]);
  let rel = m && m[1].replace(/\\/g, '/').replace(/^\.?\//, '');
  if (!rel) {
    if (/module\.exports/.test(code)) {
      const first = /['"]([a-z]+):[\w-]+['"]\s*:/.exec(code);
      rel = 'electron/modules/' + (first ? first[1] : 'custom' + Date.now()) + '.cjs';
    } else if (/export\s+default/.test(code)) {
      const id = (/\bid:\s*['"]([a-z][a-z0-9-]*)['"]/.exec(code) || [])[1];
      if (!id) fail('В коде нет id модуля (id: \'имя\').');
      rel = 'src/modules/' + id + '.js';
    } else { console.warn('• пропускаю блок без FILE и без export default / module.exports'); continue; }
  }
  const abs = path.resolve(root, rel);
  if (!abs.startsWith(root + path.sep) || rel.includes('..') || !ALLOWED.some((re) => re.test(rel))) {
    console.warn('• пропускаю «' + rel + '»: сюда автоматически писать нельзя (package.json, .github и т.п. правьте вручную)');
    continue;
  }
  plan.push({ rel, abs, code });
}
if (!plan.length) fail('Не нашёл, что добавлять.');

// — проверка синтаксиса и формы модуля до записи
const tmp = fs.mkdtempSync(path.join(os.tmpdir(), 'ghost-add-'));
for (const f of plan) {
  const ext = f.rel.endsWith('.cjs') ? '.cjs' : f.rel.endsWith('.js') || f.rel.endsWith('.mjs') ? '.mjs' : null;
  if (ext) {
    const t = path.join(tmp, 'check' + ext);
    fs.writeFileSync(t, f.code);
    try { execFileSync(process.execPath, ['--check', t], { stdio: 'pipe' }); }
    catch (e) { fail('Синтаксическая ошибка в ' + f.rel + ':\n' + String(e.stderr || e.message).split('\n').slice(0, 6).join('\n') + '\nПопросите ИИ исправить и пришлите файл полностью.'); }
  }
  if (/^src\/modules\/[^_][\w-]*\.js$/.test(f.rel)) {
    if (/^[ \t]*import\s[^\n]*['"]https?:/m.test(f.code)) fail(f.rel + ': внешние import не допускаются.');
    if (!/export\s+default/.test(f.code) || !/\bid:\s*['"][\w-]+['"]/.test(f.code) || !/\bmount\s*\(/.test(f.code)) fail(f.rel + ': не похоже на модуль (нужны export default, id, mount).');
  }
  if (/\/\/\s*\.\.\.\s*(остальн|rest|existing|без изменений|unchanged)/i.test(f.code)) fail(f.rel + ': ИИ сократил код («остальное без изменений»). Попросите прислать файл ПОЛНОСТЬЮ.');
}
fs.rmSync(tmp, { recursive: true, force: true });

// — запись
const stamp = new Date().toISOString().replace(/[:.]/g, '-');
for (const f of plan) {
  const exists = fs.existsSync(f.abs);
  console.log((dry ? '[пробный запуск] ' : '') + (exists ? '↻ заменяю ' : '＋ создаю  ') + f.rel);
  if (dry) continue;
  if (exists) {
    const b = path.join(root, '.ghost-backup', stamp, f.rel);
    fs.mkdirSync(path.dirname(b), { recursive: true });
    fs.copyFileSync(f.abs, b);
  }
  fs.mkdirSync(path.dirname(f.abs), { recursive: true });
  fs.writeFileSync(f.abs, f.code);
}
console.log('\n✓ Готово: файлов — ' + plan.length + (plan.some((f) => fs.existsSync(path.join(root, '.ghost-backup', stamp))) ? '  (старые версии: .ghost-backup/' + stamp + ')' : ''));
console.log('  Проверить: RUN.bat  |  Выпустить для всех: ./ghost.sh release');

// Быстрая проверка проекта перед выпуском: синтаксис всех файлов + правильная форма модулей.
// Запуск: npm run check
import { execFileSync } from 'node:child_process';
import fs from 'node:fs';
import path from 'node:path';

const exts = new Set(['.js', '.cjs', '.mjs']);
const dirs = ['src', 'electron', 'scripts'];
let bad = 0;

function walk(dir) {
  return fs.readdirSync(dir, { withFileTypes: true }).flatMap((e) =>
    e.isDirectory() ? walk(path.join(dir, e.name)) : [path.join(dir, e.name)]);
}

const files = [...dirs.filter(fs.existsSync).flatMap(walk), 'vite.config.js', 'electron-builder.config.cjs'].filter((f) => exts.has(path.extname(f)));

for (const f of files) {
  try {
    execFileSync(process.execPath, ['--check', f], { stdio: 'pipe' });
  } catch (e) {
    bad++;
    console.error('✗ Синтаксическая ошибка в ' + f + '\n' + String(e.stderr || e.message).split('\n').slice(0, 6).join('\n'));
  }
}

// Каждый файл в src/modules (кроме «_...») должен экспортировать модуль с id и mount.
const modDir = 'src/modules';
const ids = new Map();
for (const f of fs.readdirSync(modDir).filter((n) => n.endsWith('.js') && !n.startsWith('_'))) {
  const text = fs.readFileSync(path.join(modDir, f), 'utf8');
  if (!/export\s+default/.test(text)) { bad++; console.error('✗ ' + f + ': нет export default'); continue; }
  const id = (/\bid:\s*['"]([\w-]+)['"]/.exec(text) || [])[1];
  if (!id) { bad++; console.error('✗ ' + f + ': не найден id модуля'); continue; }
  if (!/\bmount\s*\(/.test(text)) { bad++; console.error('✗ ' + f + ': нет метода mount'); }
  if (ids.has(id)) { bad++; console.error('✗ id «' + id + '» повторяется в ' + f + ' и ' + ids.get(id)); }
  ids.set(id, f);
}

if (bad) { console.error('\nНайдено проблем: ' + bad); process.exit(1); }
console.log('✓ Проверено файлов: ' + files.length + ', модулей: ' + ids.size + ' — всё в порядке');

/** 0.2.5 → 205, 1.4.0 → 10400. Растёт вместе с версией, Android требует увеличивать versionCode. */
export function versionCode(version) {
  const [a = 0, b = 0, c = 0] = String(version).split('.').map((x) => parseInt(x, 10) || 0);
  return Math.max(1, a * 10000 + b * 100 + c);
}

function findOpenssl() {
  const cands = ['openssl'];
  if (process.platform === 'win32') {
    for (const base of [process.env.ProgramFiles, process.env['ProgramFiles(x86)'], process.env.LOCALAPPDATA && path.join(process.env.LOCALAPPDATA, 'Programs')]) {
      if (!base) continue;
      cands.push(path.join(base, 'Git', 'usr', 'bin', 'openssl.exe'), path.join(base, 'Git', 'mingw64', 'bin', 'openssl.exe'));
    }
  }
  for (const c of cands) {
    const r = spawnSync(c, ['version'], { encoding: 'utf8' });
    if (r.status === 0) return c;
  }
  return null;
}

const ssl = findOpenssl();
if (!ssl) { console.error('✗ Не нашёл openssl (он идёт вместе с Git for Windows). Установите Git и повторите.'); process.exit(1); }

// Задаёт название приложения (как оно будет подписано на рабочем столе Android и в Windows).

// «Выпустить обновление с компа»: поднимает версию, коммитит, ставит тег и пушит.
// GitHub Actions сам соберёт Windows-установщик и APK и опубликует релиз — приложения увидят обновление.
//
//   npm run release                       → 0.1.0 → 0.1.1
//   npm run release -- minor              → 0.1.1 → 0.2.0
//   npm run release -- major              → 0.2.0 → 1.0.0
//   npm run release -- 0.5.0 "Что нового" → точная версия и описание
import { execFileSync } from 'node:child_process';
import fs from 'node:fs';
import { readJson } from './lib.mjs';

const git = (...args) => execFileSync('git', args, { stdio: 'inherit' });
const gitOut = (...args) => execFileSync('git', args, { encoding: 'utf8' }).trim();
const fail = (msg) => { console.error('\n✗ ' + msg + '\n'); process.exit(1); };

const pkg = readJson('package.json');
const cfg = readJson('ghost.config.json');

// — проверки до любых изменений
try { gitOut('rev-parse', '--is-inside-work-tree'); }
catch { fail('Папка не является git-репозиторием. Запустите ghost.bat → «ПОЛНЫЙ АВТОМАТ» — он всё настроит сам.'); }
try { gitOut('remote', 'get-url', 'origin'); }
catch { fail('Не подключён GitHub. Запустите ghost.bat → «ПОЛНЫЙ АВТОМАТ» — он всё настроит сам.'); }
if (/^YOUR_/i.test(cfg.repo || '')) fail('В ghost.config.json впишите свой репозиторий в поле "repo" (например "kkkuss/ghost"), закоммитьте и повторите.');

try { execFileSync(process.execPath, ['scripts/check.mjs'], { stdio: 'inherit' }); }
catch { fail('Проверка кода не прошла — исправьте ошибки выше, выпуск отменён.'); }

// — новая версия
const arg = process.argv[2] || 'patch';
const [a, b, c] = pkg.version.split('.').map(Number);
let next;
if (/^\d+\.\d+\.\d+$/.test(arg)) next = arg;
else if (arg === 'major') next = `${a + 1}.0.0`;
else if (arg === 'minor') next = b >= 99 ? `${a + 1}.0.0` : `${a}.${b + 1}.0`;
else if (arg === 'patch') next = c >= 99 ? (b >= 99 ? `${a + 1}.0.0` : `${a}.${b + 1}.0`) : `${a}.${b}.${c + 1}`;
else fail('Не понял аргумент «' + arg + '». Используйте patch, minor, major или номер вида 1.2.3');

const note = process.argv.slice(3).join(' ').trim();
const tag = 'v' + next;
try { if (gitOut('tag', '-l', tag)) fail('Тег ' + tag + ' уже существует.'); } catch { /* нет тегов */ }

pkg.version = next;
fs.writeFileSync('package.json', JSON.stringify(pkg, null, 2) + '\n');

console.log(`\nВыпускаю ${tag}…\n`);
git('add', '-A');
git('commit', '--allow-empty', '-m', note ? `${tag}: ${note}` : `Release ${tag}`);
git('tag', tag);
git('push', 'origin', 'HEAD');
git('push', 'origin', tag);

console.log(`\n✓ Версия ${next} отправлена. Сборка идёт на GitHub: https://github.com/${cfg.repo}/actions`);
console.log('  Через 10–15 минут телефон и ПК покажут «Доступно обновление ' + next + '» — даже если этот компьютер будет выключен.');

                        Start-Process msiexec.exe -ArgumentList "/i `"$msi`" /qn /norestart" -Wait
                    }
                } catch { Say ('GitHub CLI не поставился автоматически: ' + $_.Exception.Message) 'Yellow' }
            }
            Refresh-Path
        }
        if (Has gh) { Say ('GitHub CLI: ' + ((gh --version) | Select-Object -First 1)) 'Green' } else { Say 'GitHub CLI не поставился — поставьте с cli.github.com' 'Yellow' }

        # — зависимости проекта
        Say 'Устанавливаю зависимости проекта (несколько минут)…'
        & npm install --no-audit --no-fund
        if ($LASTEXITCODE -ne 0) {
            Say 'Не вышло. Пробую через зеркало…' 'Yellow'
            $env:ELECTRON_MIRROR = 'https://npmmirror.com/mirrors/electron/'
            & npm install --no-audit --no-fund
            if ($LASTEXITCODE -ne 0) { throw 'npm install завершился с ошибкой' }
        }
    }

    # — сборка установщика
    $answer = 'n'
    if (-not $InstallOnly) { $answer = Read-Host 'Собрать установщик для Windows (.exe) сейчас? [Y/n]' }
    if ($answer -ne 'n' -and $answer -ne 'N') {
        Say 'Собираю… (первый раз — до 10 минут, скачиваются компоненты)'
        Npm-Run 'win:build'
        $exe = Get-ChildItem (Join-Path $root 'release') -Filter '*-Setup-*.exe' | Sort-Object LastWriteTime -Descending | Select-Object -First 1
        if ($exe) {
            Say ('Готово: ' + $exe.FullName) 'Green'
            $run = Read-Host 'Установить на этот компьютер сейчас? [Y/n]'
            if ($run -ne 'n' -and $run -ne 'N') { Start-Process $exe.FullName }
            else { Start-Process explorer.exe (Join-Path $root 'release') }
        }
    }

    Write-Host ''
    Say 'Всё готово. Дальше:' 'Green'
    Write-Host '  • ghost.bat — меню: полный автомат, новая функция от ИИ, запуск'
    Write-Host '  • UPDATE.bat — новая функция от ИИ одним действием'
    Write-Host '  • RUN.bat — запустить приложение из исходников'
    Write-Host '  • BUILD-EXE.bat — собрать установщик .exe'
    if ($InstallOnly) { exit 0 }
}
catch {
    Write-Host ''
    Say ('ОШИБКА: ' + $_.Exception.Message) 'Red'
    $failed = $true
}

Write-Host ''
if ($failed -or -not $InstallOnly) { Read-Host 'Нажмите Enter, чтобы закрыть окно' }
